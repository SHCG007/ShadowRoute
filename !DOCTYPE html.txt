// FOUR-FACTOR PRIORITY CALCULATION

function calculatePriority() {
  const fields = ["heat", "shade", "usage", "vulnerable"];
  const values = fields.map(id =>
    Number(document.getElementById(id).value)
  );

  const result = document.getElementById("result");

  const valid = values.every(value =>
    Number.isFinite(value) && value >= 0 && value <= 100
  );

  if (!valid) {
    result.innerHTML =
      "<h3>Invalid input</h3><p>Enter a number from 0 to 100 in every field.</p>";
    return;
  }

  const score =
    values[0] * 0.40 +
    values[1] * 0.30 +
    values[2] * 0.20 +
    values[3] * 0.10;

  let level = "Low Priority";
  let advice = "Continue monitoring this area.";

  if (score >= 70) {
    level = "High Priority";
    advice = "Consider investigating this area for cooling interventions.";
  } else if (score >= 40) {
    level = "Medium Priority";
    advice = "Consider reviewing shade and heat conditions.";
  }

  result.innerHTML = `
    <h3>${score.toFixed(1)} / 100</h3>
    <p><strong>${level}</strong></p>
    <p>${advice}</p>
  `;
}

// TEXT REPORT DOWNLOAD

function downloadFile(filename, content, type) {
  const blob = new Blob([content], { type: type });
  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");

  link.href = url;
  link.download = filename;
  link.click();

  URL.revokeObjectURL(url);
}

// CSV DATA DOWNLOAD

function downloadCSV() {
  let csv = "Area,Temperature_C,Shade_Coverage_Percent\n";

  for (let i = 0; i < areas.length; i++) {
    csv += `${areas[i]},${temperatures[i]},${shadeCoverage[i]}\n`;
  }

  downloadFile("ShadowRoute_Sample_Data.csv", csv, "text/csv");
}
